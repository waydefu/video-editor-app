const roots:string[]=[];

 const store=new ProjectStore(path.join(root,"data"));await store.initialize();
 return {root,store,source:new SourceService(store,new MediaProbe())};
}
afterEach(async()=>{await Promise.all(roots.splice(0).map(root=>rm(root,{recursive:true,force:true})))});
it("OBSERVED BUG: source import writes into new project",async()=>{
 const {root,store,source}=await setup();
 const file=path.join(root,"one.mp4");await writeFile(file,"original");
 const oldId=store.getProject().id;
 let creation:ReturnType<ProjectStore["newProject"]>|undefined;
 await source.importSelected([file],undefined,p=>{if(p.phase==="ADDING" && p.processed===0) creation=store.newProject()});
 expect(creation).toBeDefined();const next=await creation!;
 expect(next.project.id).not.toEqual(oldId);
 expect(store.getProject().id).toEqual(next.project.id);
 expect(store.getProject().sources.map(s=>s.fileName)).toEqual(["one.mp4"]);
});

import { runProcess } from "../src/main/services/process-runner";
import { ConcatRenderService } from "../src/main/services/concat-render";
import { mainRenderSelections } from "../src/shared/editing-rules";
it("OBSERVED BUG: concat export replaces a source excluded from its render list",async()=>{
 const {root,store,source}=await setup();
 const active=path.join(root,"first.mp4"), excluded=path.join(root,"excluded.mp4");
 for(const [file,color] of [[active,"red"],[excluded,"blue"]]) await runProcess("ffmpeg",["-hide_banner","-loglevel","error","-f","lavfi","-i",`color=${color}:s=320x180:r=30`,"-t","1","-c:v","libx264","-pix_fmt","yuv420p","-y",file]);
 await source.importSelected([active,excluded]);
 await Promise.all(store.getProject().sources.map(asset=>source.ensureMetadata(asset.id)));
 const excludedAsset=store.getProject().sources.find(a=>a.sourcePath===excluded)!;
 await store.removeMainAsset(excludedAsset.id);
 const before=await readFile(excluded);
 const clipSelections=mainRenderSelections(store.getProject());
 await new ConcatRenderService(store,source).render({outputToken:"test",orderedAssetIds:clipSelections.map(c=>c.assetId),clipSelections,transitionSeconds:0.3,resolution:"360P",includeWatermark:false},excluded);
 expect((await readFile(excluded)).equals(before)).toBe(false);
},30000);

it("OBSERVED BUG R2-07: automatic output created by another process during render is overwritten at finalization",async()=>{
 const {root,store,source}=await setup();
 const active=path.join(root,"first.mp4"),output=path.join(root,"new-output.mp4");
 await runProcess("ffmpeg",["-hide_banner","-loglevel","error","-f","lavfi","-i","color=red:s=320x180:r=30","-t","1","-c:v","libx264","-pix_fmt","yuv420p","-y",active]);
 await source.importSelected([active]);await source.ensureMetadata(store.getProject().sources[0].id);
 const clipSelections=mainRenderSelections(store.getProject());
 let inserted=false;
 // The callback happens synchronously before finalization. Simulate an external writer.
 const {writeFileSync}=await import("node:fs");
 await new ConcatRenderService(store,source).render({outputToken:"test",orderedAssetIds:clipSelections.map(c=>c.assetId),clipSelections,transitionSeconds:0.3,resolution:"360P",includeWatermark:false},output,undefined,progress=>{
   if(progress.phase==="FINALIZING"&&!inserted){writeFileSync(output,"other writer's file");inserted=true}
 });
 expect(inserted).toBe(true);expect((await readFile(output)).toString()).not.toBe("other writer's file");
},30000);
